		} % end scalebox

	\visible<2->{
	Recall the solutions to the voltage and current wave equations:
	%\vspace{0pt}
	\eq{
		\widetilde{V} &= V_0^+  e^{-\gamma z} +  V_0^-  e^{\gamma z} \\
		\widetilde{I} &= I_0^+  e^{-\gamma z} +  I_0^-  e^{\gamma z} 
	} % end eq
	} % end visible 2

	\end{textblock}
	
	\begin{textblock}{7.5}(8,3)
	\begin{myblock}[0.9]{Goal}
			Match \myz{s} to \myz{in}, so all \myp{s} is delivered to the load.
	\end{myblock}
	
	\visible<3->{
	\vspace{20pt}
	With, $\Gamma (z) = \Gamma_L e^{2\gamma z} = \dfrac{V_0^-}{V_0^+}e^{2\gamma z}$, we can write the \textbf{lossless case} as:
	\eq{
		\widetilde{V} &= V_0^+  e^{-j\beta z} \lp 1 + \Gamma_L e^{j2\beta z} \rp\\
		\widetilde{I} &= I_0^+  e^{-j\beta z} \lp 1 +\Gamma_L  e^{j2\beta z} \rp\\
	} % end eq
	} % end visible 3
	\end{textblock}

\end{frame}

\begin{frame}{Tx Line Equation}
	\begin{textblock}{14.5}(0.5,3)
	Now, we can express the \textbf{input impedance} (\myz{in}), at $z=-\ell$ as:
	\eq{
		\visible<1->{\myz{in} &= \dfrac{\widetilde{V}}{\widetilde{I}} = \dfrac{V_0^+  e^{-j\beta z} \lp 1 + |\Gamma_L | e^{j2\beta z} \rp}{I_0^+  e^{-j\beta z} \lp 1 +|\Gamma_L | e^{j2\beta z} \rp} 
		= \myz{0} \lb \dfrac{1 + \Gamma_L e^{-j2\beta \ell}}{1 - \Gamma_L e^{-j2\beta \ell}} \rb \\ }
		\visible<2->{&= \myz{0} \lb \dfrac{1 + \lp \dfrac{\myz{L}-\myz{0}}{\myz{L}+\myz{0}} \rp e^{-j2\beta \ell}}{1 - \lp \dfrac{\myz{L}-\myz{0}}{\myz{L}+\myz{0}} \rp e^{-j2\beta \ell}} \rb \\ }
		\visible<3->{&= 	\myz{0} \lb \dfrac{\lp \myz{L} + \myz{0} \rp e^{+j\beta \ell} + \lp \myz{L}-\myz{0} \rp e^{-j\beta \ell}}{\lp \myz{L} + \myz{0} \rp e^{+j\beta \ell} - \lp \myz{L}-\myz{0} \rp e^{-j\beta \ell}} \rb \\ }
		\visible<4->{&=\myz{0} 
		\lb 
			\dfrac{\myz{L} 2 \cosp{\beta\ell} + j\myz{0} 2\sinp{\beta\ell}}
			{\myz{0} 2 \cosp{\beta\ell} + j\myz{L} 2\sinp{\beta\ell}} 
		\rb \\
		\myz{in} &= \myz{0} 
		\lb 
			\dfrac{\myz{L}  + j\myz{0} \tanp{\beta\ell}}
			{\myz{0} + j\myz{L} \tanp{\beta\ell}} 
		\rb 
		}
	}	
	\end{textblock}
\end{frame}

\begin{frame}{Tx Line Equation}
	\begin{textblock}{8.5}(0,3)
	\begin{myblock}[0.9]{General}
		\vspace{-12pt}
		\eq{
			\myz{in} &= \myz{0} 
		\lb 
			\dfrac{\myz{L}  + j\myz{0} \text{tanh}\lp{\gamma\ell}\rp}
			{\myz{0} + j\myz{L} \text{tanh}\lp{\gamma\ell}\rp} 
		\rb 	
\\
			\myz{in} &= 
			\dfrac{z_{L}  + j \text{tanh}\lp{\gamma\ell}\rp}
			{1 + jz_{L} \text{tanh}\lp{\gamma\ell}\rp} \\
			 z_L &= \dfrac{\myz{L}}{Z_0}
		}
	\end{myblock}
	\end{textblock}
	
	\begin{textblock}{8.5}(7.75,3)
	\begin{myblock}[0.9]{Lossless}
		\vspace{-12pt}
		\eq{
			\myz{in} &= \myz{0} 
		\lb 
			\dfrac{\myz{L}  + j\myz{0} \tanp{\beta\ell}}
			{\myz{0} + j\myz{L} \tanp{\beta\ell}} 
		\rb 	
\\
			\myz{in} &= 
			\myz{0}
			\lb
				\dfrac{z_{L}  + j \tanp{\beta\ell}}
				{1 + jz_{L} \tanp{\beta\ell}} 
			\rb \\
			 z_L &= \dfrac{\myz{L}}{Z_0}
		}
	\end{myblock}
	\end{textblock}
\end{frame}

	\begin{textblock}{11}(4.75,2.75)
	\begin{myblock}[0.95]{Process}
	
	\begin{small}
	\begin{enumerate}[<+->]
	
		\item Normalize impedances to \myz{0}
		
		\item Find reflection coefficient ($\Gamma_L$)
		
		\item Holding the \handoutgap{0.75in}{magnitude} of $\Gamma_L$ constant, find the input impedance $z_{\text{in}} = \handoutgap{0.5in}{1 + jx}$ 
		
		$\implies |\Gamma_L | = \dfrac{z_{\text{in}} - 1}{z_{\text{in}} + 1}$
		
		\item Determine capacitance or inductance from $X$. 
		
		$\implies Z_C = \dfrac{1}{j\omega C} \text{ or } Z_L = j\omega L$
		
		\item Find the length of TX line to make:
		
		$\implies z_{\text{in}} = 1+jX =  \dfrac{z_L + j\tanp{\beta \ell}}{1+jz_L\tanp{\beta\ell}}$
		

	\end{enumerate}
	\end{small}
	
	\end{myblock}
	\end{textblock}

		\vspace{-8pt}
		\eq{
			|\Gamma_L | &= 0.278 = \left| \dfrac{z_{\text{in}} - 1}{z_{\text{in}} + 1} \right|\\
			&= \left| \dfrac{1 + jx -1}{1 + jx +1}\right| = \left|\dfrac{jx}{2+jx}\right| \\
			&= \dfrac{\sqrt{0 + x^2}}{\sqrt{2^2 + x^2}} \\
			0.278^2 &= \dfrac{x^2}{4 + x^2} \\
			&\implies x = \pm 0.579 \\
			&\implies X = \pm (0.579)\myz{0}  = 28.95\ohms \\
			-j28.95& = \dfrac{1}{j\omega C} \\
			+j28.95 &= j\omega L
