\begin{frame}{Uniform Plane Waves}
	\begin{textblock}{7.5}(0.5,3.5) 
	Assumptions:
	\begin{itemize}
		\item Simple Media:
		\begin{itemize}
			\setlength\itemsep{8pt}
			\item \handoutgap{1in}{Linear}
			\item \handoutgap{1in}{Isotropic}
			\item \handoutgap{1in}{Homogeneous}
			\item \handoutgap{1in}{Non-dispersive}
		\end{itemize}
		\item Magnitude and phase are constant in a plane perpendicular to direction of propagation (assume $z$)
		\visible<0|handout:1>{
		\alert{
		\eq{
			\partd{\phasor{E}}{x} = 	\partd{\phasor{E}}{y} = \partd{\phasor{H}}{x} = \partd{\phasor{H}}{y} = 0
		}  % end eq
		} % end alert
		} % end visible
	\end{itemize}
	\end{textblock}

\begin{textblock}{7.5}(8.25,2.5)
Which means:

\visible<0|handout:1>{
\alert{
\eq{
	\partd{}{x}\phasor{E}_x = \partd{}{x}\phasor{E}_y = \partd{}{x}\phasor{E}_z = 0\\
	\partd{}{x}\phasor{H}_x = \partd{}{x}\phasor{H}_y = \partd{}{x}\phasor{H}_z = 0
} % end eq
} % end alert
} % end visible

Now, the wave equation can be written as: 
\begin{small}
\visible<0|handout:1>{
\alert{
\eq{
	&\lb \partd{^2}{x^2} + \partd{^2}{y^2} + \partd{^2}{z^2} \rb 
	\lp \xhat \phasor{E}_x + \yhat \phasor{E}_y  + \zhat \phasor{E}_z \rp  \\
	&+ \gamma^2 \lp \xhat \phasor{E}_x + \yhat \phasor{E}_y  + \zhat \phasor{E}_z \rp = 0 
} % end eq
} % end alert
} % end visible
\end{small}
\end{textblock}
\end{frame}

\begin{frame}{Uniform Plane Waves: Fields}
\begin{textblock}{7.5}(0.5,3) 
Each component separately must equal zero, allowing us to write 3 equations:
\begin{small}
\visible<0|handout:1>{
\alert{
\eq{
	\lb \partd{^2}{x^2} + \partd{^2}{y^2} + \partd{^2}{z^2} \rb \phasor{E}_{x,y,z} + \gamma^2 \phasor{E}_{x,y,z} = 0 
} % end eq
} % end alert
} % end visible
\end{small}
So, the UPW conditions reduce these equations to:
\vspace{-12pt}
\begin{myblock}[0.9]{}
\begin{small}
\visible<0|handout:1>{
\alert{
\vspace{-12pt}
\eq{
	\partd{^2}{z^2}\phasor{E}_{x,y,z} + \gamma^2 \phasor{E}_{x,y,z} \\
	} % end eq
	\vspace{-12pt}
	} % end alert
	} % end visible
\end{small}
\end{myblock}
\end{textblock}


\begin{textblock}{7.5}(8.5,3)
We can make one more simplification, by considering Ampere's Law:
%\vspace{-8pt}
\begin{small}
\visible<0|handout:1>{
\alert{
\eq{
	\curl{\phasor{H}} &= \lp \sigma +  j\omega \varepsilon\rp\phasor{E}\\
	\zhat \mycdot \lp \curl{\phasor{H}} \rp &= \zhat\mycdot\lb \lp \sigma +  j\omega \varepsilon\rp\phasor{E} \rb \\
	\partd{\phasor{H}_x}{y} - \partd{\phasor{H}_y}{x}  &= \lp \sigma +  j\omega \varepsilon\rp \phasor{E}_{z} \\
	0-0 &= \lp \sigma +  j\omega \varepsilon\rp\phasor{E}_{z} \\
	0 &= \phasor{E}_{z}
} % end eq
} % end alert
} % end visible
\end{small}
%\only<1|handout:0>{\vspace{-32pt}}
\vspace{-24pt}
\begin{myblock}[0.9]{}
\begin{small}
\vspace{-12pt}
\visible<0|handout:1>{
\alert{
\eq{
	\handoutgap{0pt}{\partd{^2}{z^2}\phasor{E}_{x,y} + \gamma^2 \phasor{E}_{x,y}} 
}	% end eq
\vspace{-12pt}
} % end alert
} % end visible
\end{small}
\end{myblock}
\end{textblock}
\end{frame}

\begin{frame}{Uniform Plane Waves: Fields}
\begin{textblock}{7.5}(0.5,3.5) 
We know how to find solutions for the 2nd order ODEs (right?!)
\begin{small}
\visible<0|handout:1>{
\vspace{-12pt}
\alert{
\eq{
	&\partd{^2}{z^2}\phasor{E}_{x} + \gamma^2 \phasor{E}_{x}  = 0\\
	&\;\implies\;
	\phasor{E}_x (z) = E^{+}_{x0} e^{-j\gamma z}  + E^{-}_{x0} e^{j\gamma z} \\
	&\partd{^2}{z^2}\phasor{E}_{y} + \gamma^2 \phasor{E}_{y} = 0 \\
	&\;\implies\;
	\phasor{E}_y (z) = E^{+}_{y0} e^{-j\gamma z}  + E^{-}_{y0} e^{j\gamma z} \\
	&\phasor{E} = \xhat \phasor{E}_x + \yhat \phasor{E}_y \\
	&\phasor{H}_x (z) = H^{+}_{x0} e^{-j\gamma z}  + H^{-}_{x0} e^{j\gamma z} \\
	&\phasor{H}_y (z) = H^{+}_{y0} e^{-j\gamma z}  + H^{-}_{y0} e^{j\gamma z} \\
	&\phasor{H} = \xhat \phasor{H}_x + \yhat \phasor{H}_y\\
	&\gamma = \alpha + j\beta = \sqrt{j\omega\mu\lp\sigma + j\omega\varepsilon\rp } 
} % end eq
} % end alert
} % end visible
\end{small}

\end{textblock}


\begin{textblock}{7.5}(8.5,3.5)
Now, back to time domain...multiply by \handoutgap{1in}{$e^{j\omega t}$} and \handoutgap{1.5in}{take the real part}.

\begin{small}
\visible<0|handout:1>{
\alert{
\eq{
	E_{x,y} (z,t) &= \text{Re}\left\{ E^{+}_{x0, y0} e^{-j\beta z}e^{j\omega t} \right.\\
	&\; \left. + E^{-}_{x0,y0} e^{j\beta z}e^{j\omega t}\right\} \\
	E_{x,y} (z,t) &= E^{+}_{x0,y0} e^{-\alpha z}\cosp{\omega t - \beta z} \\
	&\; + E^{-}_{x0,y0}e^{\alpha z} \cosp{\omega t + \beta z}
} % end eq
} % end alert
} % end visible
\end{small}

\end{textblock}
\end{frame}

\begin{frame}{UPW Summary}
\begin{textblock}{8}(0.5,2)
\begin{myblock}[0.95]{Maxwell's Equations}
\vspace{-12pt}
\eq{
&\text{Gauss}\longrightarrow\div{\phasor{E}} = \dfrac{\rho_v}{\varepsilon} \\
&\text{Gauss}\longrightarrow\div{\phasor{H}} = 0 \\
&\text{Ampere}\longrightarrow\curl{\phasor{H}} = \lp \sigma + j\omega\varepsilon \rp \phasor{E} \\
&\text{Faraday}\longrightarrow\curl{\phasor{E}} = -j\omega\mu\phasor{H} 
}
\vspace{-12pt}
\end{myblock}
\end{textblock}
\begin{textblock}{6}(9,2)
\begin{myblock}[0.9]{Helmholtz}
\vspace{-12pt}
\eq{
&\nabla^2 \phasor{E} - \gamma^2 \phasor{E} = 0 \\
&\nabla^2 \phasor{H} - \gamma^2 \phasor{H} = 0 \\
&\gamma^2 = j\omega\mu\lp\sigma + j\omega\varepsilon\rp \\[-4pt]
}
\vspace{-12pt}
\end{myblock}
\end{textblock}
\begin{textblock}{14.5}(0.5,9.75)
\begin{myblock}[0.99]{Field Solutions}
\vspace{-18pt}
\eq{
&E_{x,y} (z,t) = E^{+}_{x0,y0} e^{-\alpha z}\cosp{\omega t - \beta z} 
+ E^{-}_{x0,y0} e^{\alpha z}\cosp{\omega t + \beta z}  \\
&H_{x,y} (z,t) = H^{+}_{x0,y0} e^{-\alpha z}\cosp{\omega t - \beta z} 
+ H^{-}_{x0,y0} e^{\alpha z}\cosp{\omega t + \beta z}  
}
\vspace{-18pt}
\end{myblock}
\end{textblock}
\end{frame}

	\begin{itemize}
		\item For a UPW, the electric and magnetic field \handoutgap{1in}{vectors} are \handoutgap{1in}{perpendicular} to propagation.

		\item The \handoutgap{1in}{direction} of the \handoutgap{1in}{electric field} and \handoutgap{1in}{magnetic field} are related
	\end{itemize}

\section{Characteristics}
\begin{frame}{Uniform Plane Waves: Characteristics}
\begin{textblock}{7.5}(0.5,3) 
Assume an $\xhat\;$ Electric field... 
\begin{small}
\visible<0|handout:1>{
\alert{
\eq{
	&\phasor{E} (z) = \xhat \phasor{E}_x (z) \\
	&\curl{\phasor{E}} = -j\omega\mu\phasor{H} \\
	&\xhat\lp 0 \rp  - \yhat \lp 0 - \partd{\phasor{E}_x}{z} \rp + \zhat \lp 0 - \partd{\phasor{E}_x}{y} \rp \\[8pt]%= -j\omega\mu\phasor{H} \\ 
	& \implies \yhat\partd{\phasor{E}_x}{z}
	=  -j\omega\mu\phasor{H}  \\
	&\partd{\phasor{E}_x}{z} = \partd{}{z} \lp E^{+}_{x0} e^{-\gamma z} \rp + \partd{}{z} \lp E^{-}_{x0} e^{\gamma z} \rp \\
	&\implies \yhat\lb -\gamma E^{+}_{x0} e^{-\gamma z} + \gamma E^{-}_{x0} e^{\gamma z} \rb =  -j\omega\mu\phasor{H}  \\[8pt]
	&\implies \phasor{H} = \yhat\lb \dfrac{ E^{+}_{x0}}{j\omega\mu} e^{-\gamma z} -  \dfrac{ E^{-}_{x0}}{j\omega\mu} e^{\gamma z}\rb \\[8pt]
	&\implies \phasor{H} = \yhat \lb  H^{+}_{y0}e^{-\gamma z} + H^{-}_{y0}e^{\gamma z} \rb\\
} % end eq
} % end alert
} % end visible
\end{small}
\end{textblock}

\end{textblock}


\begin{textblock}{6.5}(9,3)
Intrinsic impedance is like characteristic impedance...
\begin{small}

		\eta &= \dfrac{E^{+}_{x0}}{H^{+}_{y0}}

\section{General UPW}
\begin{frame}{Uniform Plane Waves: General}
\begin{textblock}{14.5}(0.5,3)

	&\phasor{H}(z,t)  =  \xhat {H}_x (z,t) + \yhat {H}_y (z,t) \\[4pt]

	\phasor{H} = \dfrac{1}{\eta} \hat{a}_{p} \times \phasor{E}
	&\eta = \sqrt{\dfrac{j\omega\mu}{\sigma + j\omega\varepsilon}}
}

\end{myblock}

\section{Dielectrics}
\begin{frame}{Uniform Plane Waves: Dielectrics}
\begin{textblock}{7.5}(0.5,3.5) 
For dielectrics $\lp \sigma=0, \varepsilon=\varepsilon_r \varepsilon_0, \mu=\mu_0 \rp$:
\begin{small}
\visible<0|handout:1>{
\alert{
	\eq{
		\gamma &= \sqrt{j\omega\mu_0 \lp 0 + j\omega\varepsilon_r\varepsilon_0\rp} \\
		&= \sqrt{-\omega^2 \mu_0 \varepsilon_r\varepsilon_0} \\
		&= j\omega\sqrt{\mu_0\varepsilon_r\varepsilon_0} 			
			= \dfrac{j\omega\sqrt{\varepsilon_r}}{c} \\
			&= \alpha + j\beta \\
		\eta &= \dfrac{j\omega\mu_0}{0 + j\omega\varepsilon_r\varepsilon_0} = \sqrt{\dfrac{\mu_0}{\varepsilon_r\varepsilon_0}} \\
		&= \dfrac{120\pi}{\sqrt{\varepsilon_r}} \\
		u_p &= \dfrac{\omega}{\beta} = \dfrac{\omega}{\omega\sqrt{\mu_0 \varepsilon_0\varepsilon_r}}  = \dfrac{c}{\sqrt{\varepsilon_r}}
	} % end eq
	} % end alert
} % end visible
