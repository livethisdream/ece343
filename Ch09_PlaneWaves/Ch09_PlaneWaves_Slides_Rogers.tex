\section{Phasors}
\begin{frame}{Maxwell's Equations: Phasors}
\begin{textblock}{8}(0.5,3.5)
\circled{1}\; Starting with Gauss' Law:
\begin{small}
\eq{
		\handoutgap{2in}{\div{\vec{E}} = \dfrac{\rho_v}{\varepsilon}} 
}
\end{small}
\circled{2}\; Assume that the electric and magnetic field are sinusoidally varying:
\vspace{-8pt}
\eq{
	\vec{E},\;\vec{H} = A_{e,h} \cosp{\omega t - \beta z + \phi}
}

\vspace{-8pt}

\circled{3}\; Using a phasor representation of electric flux density and charge density:

\vspace{-18pt}

\begin{small}
\eq{
	\vec{E} &= \text{Re} \left\{ \widetilde{\boldmath{E}} e^{j\omega t}  \right\} \\
	\implies \widetilde{\boldmath{E}} &=  e^{-j\lp\beta z - \phi\rp} = e^{-j\beta z}e^{j\phi} \\
	\rho_v &= \text{Re} \lbr \widetilde{\rho}_v e^{j\omega t}  \rbr
}
\end{small}
\end{textblock}
\begin{textblock}{7}(8.5,3.5)
\begin{small}
\circled{4}\; What is the phasor form of Gauss' Law?
\eq{
	&\nabla \mycdot \lb \re{\widetilde{\boldmath{E}} e^{j\omega t} } \rb = \dfrac{1}{\varepsilon} \re{\widetilde{\rho}_v e^{j\omega t}}  \\[8pt]
	&\re{\nabla\mycdot \lb \widetilde{\boldmath{E}} e^{j\omega t} \rb } = \dfrac{1}{\varepsilon}\re{\widetilde{\rho}_v e^{j\omega t}}  \\[8pt]
	&\re{\lb \div{\widetilde{\boldmath{E}} }  \rb e^{j\omega t} } = \dfrac{1}{\varepsilon}\re{\widetilde{\rho}_v e^{j\omega t}}  
}
\end{small}
\begin{myblock}[0.9]{Gauss' Law}
	\eq{
		\handoutgap{0pt}{\div{\widetilde{\boldmath{E}}} = \dfrac{\widetilde{\rho}_v}{\varepsilon}}
		}
\end{myblock}
\end{textblock}
\end{frame}

\begin{textblock}{8}(0.5,3.5)

\eq{
		\handoutgap{2in}{\curl{\vec{E}} = -\partd{\vec{B}}{t} }
}
\end{small}
\circled{2}\; Assume that the electric and magnetic field are sinusoidally varying:
\vspace{-8pt}
\eq{
	\vec{E},\;\vec{H} = A_{e,h}\cosp{\omega t - \beta z + \phi}
}

\vspace{-8pt}

\circled{3}\; Using the phasor representations: 

\vspace{-18pt}

\begin{small}
\eq{
	\vec{E} &= \re{\phasor{E} e^{j\omega t} }   \\
	\vec{H} &= \re{\phasor{H} e^{j\omega t} }   \\
}
\end{small}
\end{textblock}
\begin{textblock}{7}(8.5,3.5)
\begin{small}
\circled{4}\; What is the phasor form of Faraday's Law?
\eq{
	&\curl{\re{\phasor{E}}} = \re{-\partd{}{t}\phasor{B} e^{j\omega t}} \\[8pt]
	& \re{\lb \curl{\phasor{E}}\rb e^{j\omega t} } = \re{-\phasor{B} \partd{}{t} e^{j\omega t}}\\[8pt]
	&\re{\lb \curl{\phasor{E}}\rb e^{j\omega t} } = \re{\lb - j\omega \phasor{B}\rb e^{j\omega t}}
}
\end{small}
\begin{myblock}[0.9]{Faraday's Law}
	\eq{
		\handoutgap{0pt}{\curl{\phasor{E}} = -j\omega\phasor{B}}
		}
\end{myblock}
\end{textblock}
\end{frame}

\section{Helmholtz Wave Equations}
\begin{frame}{Wave Equations}
\begin{textblock}{14.5}(0.5,3)
The phasor form of Maxwell-Faraday and Maxwell-Ampere's equations are \textbf{coupled}, \textbf{partial differential equations}. To find a solution:
\begin{itemize}
	\item Use phasors to find time-independent expressions
	\item Decouple equations 
	\item Solve resulting 2nd order ODEs using normal methods
	\item Interpret results
\end{itemize}

\begin{myblock}[0.9]{Phasor Forms}
\eq{
	\curl{\phasor{E}} = -j\omega \mu \phasor{H} \qquad
	\curl{\phasor{H}} = \lp \sigma + j\omega \varepsilon \rp \phasor{E}
}
\end{myblock}
\end{textblock}

\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{frame}{Decoupling Maxwell's Equations}
\begin{textblock}{7.5}(0.5,3.5)
Starting with Faraday's law:
\begin{small}
\eq{
	&\curl{\phasor{E}} = -j\omega\mu \phasor{H} \\
	&\curl{\curl{\phasor{E}}} = \curl{\lp -j\omega\mu \phasor{H}\rp} \\
	&\nabla \lp \cancelto{0}{\div{\phasor{E}}}\rp - \mydyad{\nabla}^2\phasor{E} = -j\omega\mu\lp\curl{\phasor{H}}\rp \\
	&\text{Ampere:  } \curl{\phasor{H}} =  j\omega \varepsilon\phasor{E} \\[8pt]
	\implies &\handoutgap{1in}{\mydyad{\nabla}^2 \phasor{E} - \gamma^2\phasor{E} = 0}\\
	&\gamma^2 =  j\omega\mu\lp\sigma + j\omega\varepsilon \rp 
}
\end{small}
\end{textblock}

\begin{textblock}{7}(8.5,3.5)
Your turn -- decouple starting with Ampere's Law...
\begin{myblock}[0.9]{Helmholtz Wave Equations}
\begin{small}
\eq{
	\handoutgap{0pt}{\mydyad{\nabla}^2 \phasor{E} - \gamma^2\phasor{E} = 0} \\[8pt]
	\handoutgap{0pt}{\mydyad{\nabla}^2 \phasor{H} - \gamma^2\phasor{H} = 0} \\[8pt]
	\handoutgap{0pt}{\gamma^2 =  j\omega\mu\lp\sigma + j\omega\varepsilon \rp }\\[8pt]
}	
\end{small}
\end{myblock}

\end{textblock}

\end{frame}

\begin{frame}{Uniform Plane Waves}
	\begin{textblock}{7.5}(0.5,3.5) 
	Assumptions:
	\begin{itemize}
		\item Simple Media:
		\begin{itemize}
			\setlength\itemsep{8pt}
			\item \handoutgap{1in}{Linear}
			\item \handoutgap{1in}{Isotropic}
			\item \handoutgap{1in}{Homogeneous}
			\item \handoutgap{1in}{Non-dispersive}
		\end{itemize}
		\item Magnitude and phase are constant in a plane perpendicular to direction of propagation (assume $z$)
		\eq{
			\partd{\phasor{E}}{x} = 	\partd{\phasor{E}}{y} = \partd{\phasor{H}}{x} = \partd{\phasor{H}}{y} = \handoutgap{0pt}{0}
		} 
	\end{itemize}
	\end{textblock}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{textblock}{7.5}(8.5,3.5)
Which means:
\vspace{-8pt}
\eq{
	\partd{}{x}\phasor{E}_x = \partd{}{x}\phasor{E}_y = \partd{}{x}\phasor{E}_z = \handoutgap{0pt}{0} \\
	\partd{}{x}\phasor{H}_x = \partd{}{x}\phasor{H}_y = \partd{}{x}\phasor{H}_z = \handoutgap{0pt}{0}
}

Now, the wave equation can be written as: 
\begin{small}
\eq{
	&\lb \partd{^2}{x^2} + \partd{^2}{y^2} + \partd{^2}{z^2} \rb 
	\lp \xhat \phasor{E}_x + \yhat \phasor{E}_y  + \zhat \phasor{E}_z \rp  \\
	&+ \gamma^2 \lp \xhat \phasor{E}_x + \yhat \phasor{E}_y  + \zhat \phasor{E}_z \rp = 0 
}
\end{small}
\end{textblock}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Uniform Plane Waves: Fields}
\begin{textblock}{7.5}(0.5,3.5) 
Each component separately must equal zero, allowing us to write 3 equations:
\begin{small}
\eq{
	\lb \partd{^2}{x^2} + \partd{^2}{y^2} + \partd{^2}{z^2} \rb \phasor{E}_{x,y,z} + \gamma^2 \phasor{E}_{x,y,z} = 0 
}
\end{small}
...but what do our UPW conditions reduce these equations to? 

\begin{myblock}[0.9]{}
\begin{small}
\eq{
	\handoutgap{0pt}{} \\
	\handoutgap{0pt}{\partd{^2}{z^2}\phasor{E}_{x,y,z} + \gamma^2 \phasor{E}_{x,y,z}} \\
	}
\end{small}
\end{myblock}
\end{textblock}


\begin{textblock}{7.5}(8.5,3.5)
We can make one more simplification, by considering Ampere's Law:
\begin{small}
\eq{
	&\curl{\phasor{H}} = \lp \sigma +  j\omega \varepsilon\rp\phasor{E}\\
	&\zhat \mycdot \lp \curl{\phasor{H}} \rp = \zhat\mycdot\lb \lp \sigma +  j\omega \varepsilon\rp\phasor{E} \rb \\
	&\implies\handoutgap{0pt}{\partd{\phasor{H}_x}{y} - \partd{\phasor{H}_y}{x}  = \lp \sigma +  j\omega \varepsilon\rp \phasor{E}_z} \\
	&\implies \handoutgap{0pt}{0-0 = \lp \sigma +  j\omega \varepsilon\rp\phasor{E}_z} \\
	&\implies \handoutgap{0pt}{0 = \phasor{E}_z}
}
\end{small}

\begin{myblock}[0.9]{}
\begin{small}
\eq{
	\handoutgap{0pt}{\partd{^2}{z^2}\phasor{E}_{x,y} + \gamma^2 \phasor{E}_{x,y}} \\
}	
\end{small}
\end{myblock}
\end{textblock}
\end{frame}

\begin{frame}{Uniform Plane Waves: Fields}
\begin{textblock}{7.5}(0.5,3.5) 
We know how to find solutions for the 2nd order ODEs (right?!)
\begin{small}
\eq{
	&\partd{^2}{z^2}\phasor{E}_{x} + \gamma^2 \phasor{E}_{x}  = 0\\
	&\;\implies\;
	\phasor{E}_x (z) = E^{+}_{x0} e^{-j\gamma z}  + E^{-}_{x0} e^{j\gamma z} \\
	&\partd{^2}{z^2}\phasor{E}_{y} + \gamma^2 \phasor{E}_{y} = 0 \\
	&\;\implies\;
	\phasor{E}_x (z) = E^{+}_{y0} e^{-j\gamma z}  + E^{-}_{y0} e^{j\gamma z} \\
	&\phasor{E} = \xhat \phasor{E}_x + \yhat \phasor{E}_y \\[12pt]
	&\phasor{H}_x (z) = H^{+}_{x0} e^{-j\gamma z}  + H^{-}_{x0} e^{j\gamma z} \\
	&\phasor{H}_y (z) = H^{+}_{y0} e^{-j\gamma z}  + H^{-}_{y0} e^{j\gamma z} \\
	&\phasor{H} = \xhat \phasor{H}_x + \yhat \phasor{H}_y
}
\end{small}
Do these look familiar?!?!  
\end{textblock}

Now, back to time domain...multiply by \handoutgap{1in}{$e^{j\omega t}$} and \handoutgap{1.5in}{take the real part}.

Also, remember...

\begin{small}
\eq{

	E_x (z,t) &= \text{Re}\left\{ E^{+}_{x0} e^{-j\beta z}\handoutgap{0.5in}{e^{j\omega t}} \right.\\[12pt]
	&\qquad\quad \left. + E^{-}_{x0} e^{j\beta z}\handoutgap{0.5in}{e^{j\omega t}}\right\} \\
	&\qquad\qquad\vdots \\
	E_x (z,t) &= \handoutgap{0pt}{E^{+}_{x0} e^{-\alpha z}\cosp{\omega t - \beta z} } \\
	&\handoutgap{0pt}{+ E^{-}_{x0}e^{\alpha z} \cosp{\omega t + \beta z}}
}
\end{small}

\end{textblock}
\end{frame}

\begin{frame}{Uniform Plane Waves: Interpretation}
\begin{textblock}{14.5}(0.5,2.25)

\begin{myblock}[0.98]{Observations}
	\begin{itemize}

\section{Characteristics}
\begin{frame}{Uniform Plane Waves: Characteristics}
\begin{textblock}{7.5}(0.5,3) 
Assume an $\xhat\;$ Electric field... 
\begin{small}
\eq{
	&\phasor{E} (z) = \xhat \phasor{E}_x (z) \\
	&\curl{\phasor{E}} = -j\omega\mu\phasor{H} \\
	&\xhat\lp 0 \rp  - \yhat \lp 0 - \partd{\phasor{E}_x}{z} \rp + \zhat \lp 0 - \partd{\phasor{E}_x}{y} \rp \\[8pt]%= -j\omega\mu\phasor{H} \\ 
	& \implies \handoutgap{1in}{\yhat\partd{\phasor{E}_x}{z}} 
	=  -j\omega\mu\phasor{H}  \\
	&\partd{\phasor{E}_x}{z} = \partd{}{z} \lp E^{+}_{x0} e^{-\gamma z} \rp + \partd{}{z} \lp E^{-}_{x0} e^{\gamma z} \rp \\
	&\implies \yhat\lb -\gamma E^{+}_{x0} e^{-\gamma z} + \gamma E^{-}_{x0} e^{\gamma z} \rb =  -j\omega\mu\phasor{H}  \\[8pt]
	&\implies \phasor{H} = \yhat\lb \handoutgap{0.5in}{\dfrac{ E^{+}_{x0}}{j\omega\mu}} e^{-\gamma z} -  \handoutgap{0.5in}{\dfrac{ E^{-}_{x0}}{j\omega\mu}} e^{\gamma z}\rb \\[8pt]
	&\implies \phasor{H} = \yhat \lb  H^{+}_{y0}e^{-\gamma z} + H^{-}_{y0}e^{\gamma z} \rb\\
} % end eq
\end{small}
\end{textblock}


\begin{textblock}{0.5}(8.75,3)
	{\color{Blue4}{\rule{1pt}{2.8in}}}
\end{textblock}


\begin{textblock}{6}(9,3.5)
Intrinsic impedance is like characteristic impedance...
\begin{small}
\vspace{-8pt}
	\eq{
		\eta &= \dfrac{E^{+}_{x0}}{H^{+}_{y0}} 
			= \dfrac{E^{+}_{x0}}{\frac{E^{+}_{x0}}{j\omega\mu}} \\
		 &= \dfrac{j\omega\mu}{\gamma} 
		 = \dfrac{j\omega\mu}{\sqrt{j\omega\mu\lp\sigma + j\omega\varepsilon\rp}} \\[8pt]
		 &\eta = \handoutgap{1in}{\sqrt{\dfrac{j\omega\mu}{\sigma + j\omega\varepsilon}}}
	}	
\end{small}

Finally, $\phasor{E}$, $\phasor{H}$, and propagation ($\hat{a}_p$) are \textbf{generally} related:
\begin{small}
\vspace{-8pt}
\eq{
	&\phasor{E}  = -\eta \hat{a}_{p} \times \phasor{H} \\
	&\phasor{H} = \dfrac{1}{\eta} \hat{a}_{p} \times \phasor{E}
}
\end{small}

\begin{frame}{Normal Incidence (TEM): BC's}
\begin{textblock}{7.5}(0.5,3) 
Now, use boundary conditions, assuming we know incident E-field
\vspace{-12pt}
\begin{small}
\eq{
	&\vec{E}_{t1} = \vec{E}_{t2} \\
	&\implies E_{x0}^{i} + E_{x0}^{r} = E_{x0}^{t} \\
	&\hat{n}_{21} \times \lp \vec{H}_1 - \vec{H}_2 \rp = \cancelto{0}{\vec{J}_S}\\
	&\implies -\zhat \times \lp \yhat \dfrac{E_{x0}^i}{\eta_1}  + \yhat \dfrac{E_{x0}^r}{\eta_1} - \yhat \dfrac{E_{x0}^t}{\eta_2} \rp
}
\end{small}
2 equations, 2 unknowns ($E_{x0}^t$ and $E_{x0}^r$). 
\vspace{-12pt}
\begin{small}
\eq{
	&\Gamma = \dfrac{E_{x0}^r}{E_{x0}^i} = \ldots = \dfrac{\eta_2 - \eta_1}{\eta_2 + \eta_1} \\
	& T = \ldots =  \dfrac{E_{x0}^t}{E_{x0}^i}  = 1 + \Gamma  = \dfrac{2\eta_2}{\eta_2 + \eta_1}
}
\end{small}
\end{textblock}

\begin{textblock}{7.5}(8.75,3.5)
For a dielectric, $\sigma = 0$, leading to: 
\begin{small}
	\eq{
		&\eta_{k} = 120\pi \sqrt{\dfrac{\mu_{r,k}}{\varepsilon_{r,k}}} \\
		&\gamma = \sqrt{j\omega\mu \lp \sigma + j\omega\varepsilon \rp}
