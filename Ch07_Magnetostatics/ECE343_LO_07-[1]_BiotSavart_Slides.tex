\begin{frame}{Biot Savart: Line Charge}
\begin{textblock}{7}(0.5,3)
	\eq{
		&\circled{6} \text{ Integrate!}\qquad\qquad \\
		&\handoutgap{0pt}{ \vec{H} = \int_{-a}^{a}\dfrac{I \lp \phihat b dz\rp}{4\pi \lp b^2 + z^2 \rp^{3/2}}}\\
		&\handoutgap{0pt}{\qquad = \phihat \dfrac{Ib}{4\pi} \lb \dfrac{z}{b^2 \lp b^2 + z^2 \rp^{1/2}} \rb\Bigg|_{-a}^{a} }\\
		&\handoutgap{0pt}{\qquad = \phihat \dfrac{Ib}{4\pi} \lb \dfrac{a}{b^2 \lp b^2 + a^2 \rp^{1/2}} \right.}\\
		&\handoutgap{0pt}{\qquad\; \left.  - \dfrac{-a}{b^2 \lp b^2 + (-a)^2 \rp^{1/2}} \rb }\\
		&\handoutgap{0pt}{\qquad =  \phihat \dfrac{Ia}{2\pi b \lp b^2 + a^2 \rp^{1/2}} \longrightarrow \text{ ``finite'' wire} }
	}
\end{textblock}
\begin{textblock}{7}(8,3)
\eq{
	&\text{ for ``infinite'' wire } \; \implies a >> b \\
	& \handoutgap{0pt}{\vec{H} = \phihat \lim_{a>>b} \dfrac{Ia}{2\pi b \lp b^2 + a^2 \rp^{1/2}}} \\[12pt]
	& \handoutgap{0pt}{=\phihat \dfrac{I}{2\pi b \lp a^2 \rp^{1/2} }}\\[12pt]
	& \handoutgap{0pt}{=\phihat \dfrac{I}{2\pi b} }
}
\end{textblock}
\end{frame}
