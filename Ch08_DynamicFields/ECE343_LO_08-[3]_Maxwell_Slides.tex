\section{Example \#1} 
\begin{frame}{Maxwell's Equations: Example \#1}
\begin{textblock}{7}(0.5,3)
A wire loop of radius $a=10$cm lies in the $x-y$ plane (i.e., $z=0$) and contains a small gap with 1k\ohms resistor. A time-varying magnetic flux density, $B(t)$ exists, with $t_0 = 3$s and $B_0 = 0.2$T.  Determine the potential and current across the resistor. 
\eq{
	B(t) =
	\begin{cases}
		0 & t < 0 \\
		\dfrac{B_0}{t_0} & 0 \leq t \leq t_0 \\
		B_0	& t > t_0
	\end{cases} 
}
\end{textblock}
