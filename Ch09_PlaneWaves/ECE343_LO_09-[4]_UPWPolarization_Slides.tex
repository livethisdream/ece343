\begin{textblock}{14.5}(0.5,3) 
\begin{myblock}[0.95]{Definition}
Polarization is the \handoutgap{1in}{orientation} of the electric field vector.

\handoutgap{1in}{Time-harmonic} behavior affects polarization.
\end{myblock}

\section{Example 1}
\begin{frame}{Uniform Plane Waves: Polarization \# 1}
\begin{textblock}{8}(0.5,2) 
\begin{myblock}[1]{}
\begin{small}
\vspace{-12pt}
\eq{
	&\vec{E}(z,t) = \xhat E_{x0} \cosp{\omega t- \beta z} \\[-6pt]
	&\text{We know: }  \omega = 2\pi f = \dfrac{2\pi}{T} \\[-8pt]
	&\text{ Assume: } z = 0 \\[-8pt]
	&\vec{E}(0,0) = \xhat E_{x0}\cosp{0}
	 = \handoutgap{0.5in}{\xhat E_{x0}} \\
	&\vec{E}(0,\frac{T}{4}) = \xhat E_{x0}\cosp{\frac{2\pi}{T}\mycdot\frac{T}{4}}
	 = \handoutgap{0.5in}{0} \\
	 &\vec{E}(0,\frac{T}{2}) = \xhat E_{x0}\cosp{\frac{2\pi}{T}\mycdot\frac{T}{2}}
	 = \handoutgap{0.5in}{-\xhat E_{x0}} \\
	&\vec{E}(0,\frac{3T}{4}) = \xhat E_{x0}\cosp{\frac{2\pi}{T}\mycdot\frac{3T}{4}}
	 = \handoutgap{0.5in}{0} \\
	&\vec{E}(0,T) = \xhat E_{x0}\cosp{\frac{2\pi}{T}\mycdot T}
	 = \handoutgap{0.5in}{\xhat E_{x0}} 
}
\vspace{-12pt}
\end{small}
\end{myblock}
\end{textblock}
