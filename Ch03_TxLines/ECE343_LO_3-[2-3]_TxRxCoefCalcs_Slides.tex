	\begin{textblock}{8}(6.5,3)
	
	\eq{
		\visible<1->{\text{Recall} &\longrightarrow \widetilde{V}(z) = V_0^+ e^{-\gamma z} + V_0^- e^{\gamma z} \\}
		\visible<2->{\Gamma (z) &= \dfrac{V_0^-  e^{-\gamma z}}{V_0^+ e^{-\gamma z}} = \dfrac{V_0^-}{V_0^+} e^{2\gamma z} \\ }
		\visible<3->{\Gamma (z) &= \handoutgap{1in}{\left| \Gamma_L \right|e^{2\gamma z} }\\}	
		\visible<4->{\text{Lossless}\longrightarrow \gamma &= \handoutgap{1in}{j\beta}\\}	
	}

	
	\end{textblock}

	\begin{textblock}{8}(8,2.5)
	\begin{myblock}[0.9]{Wave Equations}
	\vspace{-12pt}
	\begin{small}
	\eq{
		V(z,t) &= V_0^+ e^{-\gamma z}  + V_0^- e^{\gamma z}\\
		&=  V_0^+ e^{-\alpha z} \cosp{\omega t - \beta z} \\
		&\quad + V_0^-  e^{\alpha z} \cosp{\omega t + \beta z} \\
		v_p &= \dfrac{\omega}{\beta} \quad
		\lambda  = \dfrac{2\pi}{\beta} 
	}
	\end{small}
	\end{myblock}
	
	\end{textblock}
