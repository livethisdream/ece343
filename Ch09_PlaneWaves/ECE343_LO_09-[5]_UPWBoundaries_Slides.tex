\begin{frame}{Normal Incidence (TEM): BC's}
\begin{textblock}{7.5}(0.5,3) 
Use boundary conditions, assuming we know incident E-field
\vspace{-8pt}
\begin{small}
\visible<0|handout:1>{
\alert{
\eq{
	\vec{E}_{t1} &= \vec{E}_{t2} \\
	E_{y0}^{i} + E_{y0}^{r} &= E_{y0}^{t} \\
	\hat{n}_{21} \times \lp \vec{H}_1 - \vec{H}_2 \rp &= \cancelto{0}{\vec{J}_S}\\
	 -\zhat \times \lp \yhat \dfrac{E_{y0}^i}{\eta_1}  + \yhat \dfrac{E_{y0}^r}{\eta_1} - \yhat \dfrac{E_{y0}^t}{\eta_2} \rp &=  0
} % end eq
} % end alert
} % end visible
\end{small}
2 eqns, 2 unkns ($E_{x0}^t$ \& $E_{x0}^r$):
\vspace{-8pt}
\begin{small}
\visible<0|handout:1>{
\alert{
\eq{
	&\Gamma = \dfrac{E_{x0}^r}{E_{x0}^i} = \ldots = \dfrac{\eta_2 - \eta_1}{\eta_2 + \eta_1} \\
	& T = \ldots =  \dfrac{E_{x0}^t}{E_{x0}^i}  = 1 + \Gamma  = \dfrac{2\eta_2}{\eta_2 + \eta_1}
} % end eq
} % end alert
} % end visible
\end{small}
\end{textblock}

\begin{textblock}{7.5}(8.75,3.5)
Now, for dielectric ($\sigma = 0$):
\begin{small}
\visible<0|handout:1>{
\alert{
	\eq{
		\eta_{k} &= 120\pi \sqrt{\dfrac{\mu_{r,k}}{\varepsilon_{r,k}}} \\
		\gamma &= \sqrt{j\omega\mu \lp \sigma + j\omega\varepsilon \rp} 
		= j\omega \sqrt{\mu\varepsilon}  \\
		&= \alpha + j\beta \\
		 \alpha &= 0 \qquad \beta = \omega \sqrt{\mu\varepsilon}
	} % end eq
	} % end alert
	} % end visible
\end{small}

\end{textblock}

\end{frame}

\begin{frame}{Oblique Incidence (TE)}
\begin{textblock}{7.5}(0.5,3) 
Use transverse (E-field) BCs to determine $\theta^r$ and $\theta^t$:
\begin{small}
\visible<0|handout:1>{
\alert{
\eq{
	\phasor{E}_{t1} (z=0,t) &= \phasor{E}_{t2} (z=0,t)\\
	\yhat E_{y0}^{i} e^{-j\beta_1 \lp x\sinp{\theta^i } \rp}  
	+ \yhat E_{y0}^{r} e^{-j\beta_1 \lp x\sinp{\theta^r} \rp} 
	&= \yhat E_{y0}^{t} e^{-j\beta_1 \lp x\sinp{\theta^t } \rp} \\
	\text{``Phase Matching''} &\implies \\
	-j\beta_1 x\sinp{\theta^i } &= -j\beta_1  x\sinp{\theta^r} = -j\beta_2 x\sinp{\theta^t} \\
	\theta^i &= \theta^r \\
	\beta_1 \sinp{\theta^i} &= \beta_2 \sinp{\theta^p} \\
	\theta^t &= \text{sin}^{-1}\lb \dfrac{\beta_1}{\beta_2}\sinp{\theta^i} \rb \\[1in]
	 &= \text{sin}^{-1}\lb \sqrt{ \dfrac{\varepsilon_{r1}}{\varepsilon_{r2}}}\sinp{\theta^i} \rb \\
	\theta_c &= \text{sin}^{-1}\lp \sqrt{\dfrac{\varepsilon_{r2}}{\varepsilon_{r1}}}\rp 
} % end eq
} % end alert
} % end visible
\end{small}
\end{textblock}
