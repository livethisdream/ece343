\begin{frame}{Conductivity}
\begin{textblock}{14.5}(0.5,2)
\begin{myblock}[0.9]{Definition}
\vspace{-16pt}
	\eq{
		&\vec{J} = \sigma \vec{E} \\
		&\implies \sigma \longrightarrow \dfrac{1}{\ohms} = \text{S (Siemens)}
	}
\vspace{-18pt}
\end{myblock}
\vspace{-18pt}
\begin{myblock}[0.9]{Types of Conductors}
	\begin{enumerate}
		\item ``Good'' conductor $\longrightarrow \sigma > 10^5$ S/m
		\item Semiconductor $\longrightarrow\ee{-4} < \sigma < 10$ S/m
		\item Poor insulators $\longrightarrow 10$ S/m
		\item Good insulators $\longrightarrow \sigma < \ee{-10}$ S/m
		\item Vacuum $\longrightarrow 0$
	\end{enumerate}
\end{myblock}
\end{textblock}
\end{frame}

\begin{textblock}{7}(8,3)
\only<0|handout:1>{
\alert{
\eq{
	V &= -\int \vec{E}\mycdot \vec{d\ell} \\
	  &= -\int \lp -\zhat E_z \rp\mycdot \zhat dz\\
	  &= E_z \ell \\
	  \implies \vec{J} &= \sigma \vec{E} = \zhat \sigma E_z \\
	  \implies I &= \int_{\rho=0}^{a} \int_{\phi=0}^{2\pi} \zhat \sigma E_z \mycdot \zhat \rho d\rho d\phi\\
	  &= \sigma E_z \pi a^2 \\
	  \implies R&= \dfrac{V}{I}  = \dfrac{E_z \ell}{\sigma E_z \pi a^2} = \dfrac{\ell}{\sigma A}
}
} % end alert
} % end handout

	V &= -\int_b^a \dfrac{Q}{2\pi\varepsilon_0 \rho L}  \\
		&= \dfrac{Q}{2\pi\varepsilon_0 \rho L} \text{ln}\lp \dfrac{b}{a}\rp\\

	&= \dfrac{1}{2\pi \sigma L}  \text{ln}\lp \dfrac{b}{a}\rp \\
