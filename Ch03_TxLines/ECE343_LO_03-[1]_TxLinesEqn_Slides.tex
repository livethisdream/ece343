\begin{frame}{Lumped Element Model}
	\begin{textblock}{14.5}(0.5,2.5)
		\begin{myblock}[0.9]{Telegrapher's Equations}
			\vspace{-8pt}
			\eq{
				-\partd{V(z,t)}{z} &= R^{\prime} i(z,t) + L^{\prime} \partd{i(z,t)}{t} \\
				-\partd{i(z,t)}{z} &= G^{\prime} V(z,t) + C^{\prime} \partd{V(z,t)}{t} 
			}
		\end{myblock}
	
	\only<2->{
	\begin{myblock}[0.9]{Phasor form of solutions}
	\vspace{-8pt}
			\eq{
				V(z,t) = \text{Re} \lb \widetilde{V}(z) e^{j\omega t} \rb 
				&\quad&
				i(z,t) = \text{Re} \lb \widetilde{I}(z) e^{j\omega t} \rb  \\
				\partd{V(z,t)}{t} = \partd{}{t} \text{Re} \lb \widetilde{V}(z) e^{j\omega t} \rb 
				&\quad&
				\partd{i(z,t)}{t} = \partd{}{t} \text{Re} \lb \widetilde{I}(z) e^{j\omega t} \rb \\
				= \text{Re} \lb j\omega \widetilde{V}(z) e^{j\omega t} \rb 
				&\quad& 
				= \text{Re} \lb j\omega \widetilde{I}(z) e^{j\omega t} \rb 
			}
		\end{myblock}
	} % end only 2
	\end{textblock}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{frame}{Decouple Telegrapher's Equations}
	\begin{textblock}{14.5}(0.5,3)
	Using Phasor form of solutions:
	\vspace{-12pt}
	\eqn{
		-\partd{\widetilde{V}(z)}{z} & = R^{\prime} \widetilde{I} (z) + j\omega L^{\prime} \widetilde{I}(z)  
		= \lb R^{\prime} + j\omega L^{\prime}\rb \widetilde{I}(z) \label{eq: t1}\\
			-\partd{\widetilde{I}(z)}{z} & = G^{\prime} \widetilde{V} (z) + j\omega C^{\prime} \widetilde{V}(z) 
			 = \lb G^{\prime} + j\omega C^{\prime}\rb \widetilde{V}(z) \label{eq: t2}
	} % end eqn 
	
	\vspace{-12pt}
	\visible<2->{Take $\partd{}{z}$ of (\ref{eq: t1})  to get:
	\vspace{-16pt}
		\eqn{
			-\partd{^2}{z^2} \widetilde{V}(z) &= \lb R^{\prime} + j\omega L^{\prime} \rb \partd{\widetilde{I}}{z} 
			\label{eq: t3}
		}
	} % end visible 2

	\vspace{-12pt}
	\visible<3->{
	Now, plug (\ref{eq: t2}) into (\ref{eq: t3}):
	\vspace{-12pt}
		\eqn{
			-\partd{^2}{z^2} \widetilde{V}(z) &= -\lb R^{\prime} + j\omega L^{\prime} \rb \lb G^{\prime} + j\omega C^{\prime} \rb \widetilde{V}(z)  \nn\\
			&\implies \partd{^2}{z^2} \widetilde{V}(z) - \lb R^{\prime} + j\omega L^{\prime} \rb \lb G^{\prime} + j\omega C^{\prime} \rb \widetilde{V}(z)  = 0  \nn\\
			&\implies \partd{^2}{z^2} \widetilde{V}(z) - \gamma^2 \widetilde{V}(z)  = 0 
			\text{  with: } \gamma = \sqrt{\lb R^{\prime} + j\omega L^{\prime} \rb \lb G^{\prime} + j\omega C^{\prime} \rb } \nn
		}
	} % end visible 3
	\end{textblock}
\end{frame}

\begin{frame}{Lumped Element Model: Wave Equations}
	\begin{textblock}{14.5}(0.5,3)
		\begin{myblock}[0.9]{Wave Equations}
			\vspace{-8pt}
			\eq{
				\partd{^2}{z^2} &\widetilde{V}(z) - \gamma^2 \widetilde{V}(z)  = 0 \\
				\partd{^2}{z^2} &\widetilde{I}(z) - \gamma^2 \widetilde{I}(z)  = 0  \\
				\gamma &=  \sqrt{\lb R^{\prime} + j\omega L^{\prime} \rb \lb G^{\prime} + j\omega C^{\prime} \rb}
			}
		\end{myblock}
	\end{textblock}
\end{frame}

\setbeamercovered{invisible}
\begin{frame}{Solving Wave Equations}
	\begin{textblock}{14.5}(0.5,2)
	\begin{myblock}[0.9]{Wave Equations}
			\vspace{-8pt}
			\eq{
				\partd{^2}{z^2} &\widetilde{V}(z) - \gamma^2 \widetilde{V}(z)  = 0 \\
				\partd{^2}{z^2} &\widetilde{I}(z) - \gamma^2 \widetilde{I}(z)  = 0  \\
				\gamma &=  \sqrt{\lb R^{\prime} + j\omega L^{\prime} \rb \lb G^{\prime} + j\omega C^{\prime} \rb}
				 = \alpha + j\beta 
			}
		\end{myblock}
		
		\begin{myblock}[0.9]{Solution Method}
			\begin{enumerate}[<+->]
%				\setlength\itemsep{12pt}
				\item Assume form of solution: \handoutgap{1in}{$A e^{\lambda z} $}
				\item Find characteristic equation \& it's \handoutgap{1in}{roots}
				\item Total solution is the sum of all possible solutions
				\item Enforce \handoutgap{1in}{boundary conditions} \&  \handoutgap{1in}{initial conditions}
			\end{enumerate}
		\end{myblock}
	\end{textblock}
\end{frame}

				\gamma &=   \handoutgap{0in}{\sqrt{\lb R^{\prime} + j\omega L^{\prime} \rb \lb G^{\prime} + j\omega C^{\prime} \rb} }

		\end{myblock}
		

	\end{textblock}

\begin{frame}{Back to Time Domain}
	\begin{textblock}{14.5}(0.5,3.5)
	To find the time domain solution, multiply by \handoutgap{0.5in}{$e^{j\omega t}$} and \handoutgap{0.5in}{take the real part}
	\visible<+->{
		\vspace{-8pt}
		\alert{
		\eq{
			V(z,t) &= \text{Re}\lb \widetilde{V}(z) e^{j\omega t} \rb = \text{Re} \lb V^{+}_0 e^{-\gamma z}e^{j\omega t} + V^{-}_0 e^{\gamma z}e^{j\omega t} \rb
		} % end eq
		} % end alert
	} % end visible 
	
	\vspace{-12pt}
	
	\visible<+->{
		Remember, $\gamma =$ \handoutgap{1in}{$\sqrt{\lb R^{\prime} + j\omega L^{\prime} \rb \lb G^{\prime} + j\omega C^{\prime} \rb} = \alpha + j\beta $}, which leads to:
	} % end visible

	\visible<+->{
		\vspace{-12pt}
		\alert{
		\eq{
			V(z,t) &= \text{Re} \lb V^{+}_0 e^{-\gamma z}e^{j\omega t} + V^{-}_0 e^{\gamma z}e^{j\omega t} \rb \\
				 &= \text{Re} \lb V^{+}_0 e^{-(\alpha + j\beta) z}e^{j\omega t} + V^{-}_0 e^{(\alpha + j\beta)z}e^{j\omega t} \rb \\				 
				&= \text{Re} \lb V^{+}_0 e^{-\alpha z}e^{j(\omega t -\beta z)}  V^{-}_0 e^{\alpha z}e^{j(\omega t +\beta z)} \rb
		} % end eq
		} % end alert
	} % end visible 

		\visible<+->{
		\vspace{-24pt}
		\begin{myblock}[0.9]{Instantaneous Solution to Wave Equations}
		\vspace{-12pt}
			\eq{
			V(z,t) &= V^{+}_0 e^{-\alpha z}\cosp{\omega t - \beta z} +  V^{-}_0 e^{\alpha z}\cosp{\omega t + \beta z}
			}
		\end{myblock}
		} % end visible
	\end{textblock}
\end{frame}

		\item Wavelength \handoutgap{1in}{$\lambda = \dfrac{2\pi}{\beta}$}
	
		\only<0|handout:1>{
		One period is $2\pi$ radians, so at $t=0$:
		\vspace{-12pt}
			\eq{
				\omega (0)- \beta z &= 2\pi \\
				z = \dfrac{2\pi}{\beta} = \lambda 
			} % end eq
		} % end only  
		
		\item Speed = \handoutgap{1in}{$\dfrac{\omega}{\beta}$}
	
		\only<0|handout:1>{
		Pick a point and ride the wave:
		\vspace{-12pt}
			\eq{
				\omega t - \beta z &= \text{constant} \\
				\omega t = \beta z \implies z &= \dfrac{\omega t}{\beta} \\
				u_p = v_p = \partd{z}{t} &= \dfrac{\omega}{\beta} \\[-24pt]
			} % end eq
		} % end only 
	\end{itemize}
	
	\end{textblock}

	\begin{myblock}[0.9]{Lossless/low-loss Conditions}
		\vspace{-12pt}
		\eq{
			&\gamma = \sqrt{\lp R^{\prime} + j\omega C^{\prime}\rp\lp G^{\prime} + j\omega L^{\prime} \rp} \\
			&R^{\prime} \longrightarrow G^{\prime} \text{ or: } R^{\prime}<< \omega C^{\prime} \text{ and } 
			G^{\prime} << \omega L^{\prime}\\
			&\implies \gamma = \handoutgap{1.5in}{j\omega \sqrt{L^{\prime}C^{\prime}}} \\
			&\implies \alpha = \handoutgap{1in}{0} \quad \beta = \handoutgap{1in}{\omega\sqrt{L^{\prime}C^{\prime}}}
		}
	\end{myblock}
